cmake_minimum_required(VERSION 3.21)
set_property(GLOBAL PROPERTY USE_FOLDERS ON)
set(MMFG_COMPONENTS metamod asext ascurl asqcvar asusermsg fallguys slowdl gsphysfix)
option(FORMAT_VALIDATION_ONLY "Configure formatting without compilers or native dependencies" OFF)
if(FORMAT_VALIDATION_ONLY)
    project(MetamodFallguysFormat LANGUAGES NONE)
    include(cmake/FormatValidation.cmake)
    return()
endif()
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/metamod/cmake/Platform.cmake")
    message(FATAL_ERROR "Initialize components with git submodule update --init --recursive.")
endif()
include(metamod/cmake/Platform.cmake)
project(metamod-fallguys LANGUAGES C CXX)
if(NOT CMAKE_CONFIGURATION_TYPES AND NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build configuration" FORCE)
endif()
include(metamod/cmake/Build.cmake)
mmfg_configure_install_layout()

# Explicit overrides win; otherwise inject one local tree per dependency.
foreach(dependency METAMOD ASEXT ANGELSCRIPT CAPSTONE PROCMAP FORMAT_VALIDATION)
    if(NOT DEFINED ${dependency}_SOURCE_PATH)
        set(${dependency}_SOURCE_PATH "$ENV{${dependency}_SOURCE_PATH}" CACHE PATH "External ${dependency} source override")
    endif()
endforeach()
if(NOT METAMOD_SOURCE_PATH)
    set(METAMOD_SOURCE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/metamod")
endif()
if(NOT ASEXT_SOURCE_PATH)
    set(ASEXT_SOURCE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/asext")
endif()
if(NOT ANGELSCRIPT_SOURCE_PATH)
    set(ANGELSCRIPT_SOURCE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/thirdparty/angelscript-sdk")
endif()
if(NOT CAPSTONE_SOURCE_PATH)
    set(CAPSTONE_SOURCE_PATH "${METAMOD_SOURCE_PATH}/thirdparty/capstone_fork")
endif()
if(NOT PROCMAP_SOURCE_PATH)
    set(PROCMAP_SOURCE_PATH "${METAMOD_SOURCE_PATH}/thirdparty/procmap_fork")
endif()
if(NOT FORMAT_VALIDATION_SOURCE_PATH)
    set(FORMAT_VALIDATION_SOURCE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/thirdparty/FormatValidation")
endif()
foreach(component IN LISTS MMFG_COMPONENTS)
    string(TOUPPER "${component}" upper)
    option(MMFG_BUILD_${upper} "Build ${component}" ON)
    if(MMFG_BUILD_${upper})
        add_subdirectory("${component}")
    endif()
endforeach()
include(cmake/FormatValidation.cmake)
include(cmake/TargetFolders.cmake)
mmfg_assign_solution_folders()
install(FILES build/addons/metamod/plugins.ini DESTINATION addons/metamod)
